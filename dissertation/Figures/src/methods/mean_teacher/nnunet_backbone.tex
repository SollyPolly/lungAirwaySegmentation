% Dataset126 3-D full-resolution PlainConvUNet, drawn in the PlotNeuralNet grammar
% and following the structure of that project's examples/Unet/Unet.tex:
%
%   * a stage is a RightBandedBox whose banded sub-boxes are its individual
%     convolutions, labelled underneath with their output channel count;
%   * block HEIGHT and DEPTH track spatial resolution, block WIDTH tracks channels;
%   * a narrow red slab marks each resolution reduction and a narrow blue slab each
%     transposed-convolution expansion;
%   * a translucent grey banded box is the concatenated encoder tensor;
%   * skips leave the encoder stage's north-east corner, run above the panel, and
%     drop into the matching concatenation box.
%
% Channel widths: 32 -> 2, 64 -> 2.8, 128 -> 3.6, 256 -> 4.6, 320 -> 5.5.
% Heights/depths: 40, 32, 26, 19, 13, 8 for the six resolutions.
\def\ConvColor{rgb:yellow,5;red,2.5;white,5}
\def\ConvReluColor{rgb:yellow,5;red,5;white,5}
\def\PoolColor{rgb:red,1;black,0.3}
\def\UnpoolColor{rgb:blue,2;green,1;black,0.3}
\def\SoftmaxColor{rgb:magenta,5;black,7}
\def\CatColor{rgb:white,1;black,3}
\def\CatBandColor{rgb:white,1;black,2}

\usetikzlibrary{calc}

% Spatial tensor sizes are placed as free nodes rather than through the pic's own
% zlabel.  The vendored depth label uses text width = 14*depth, so a long size
% string wraps into an unreadable column on the small deep blocks; a free node is
% independent of block size and sits on the same near-face edge as the channel
% labels.  #1 pic name, #2 label.
\newcommand{\SpatialLabel}[2]{%
  \node[font=\fontsize{15}{17}\selectfont,anchor=north,inner sep=0pt]
    at ($(#1-nearsouthwest)!0.5!(#1-nearsoutheast)+(0,-.68)$) {#2};}
